#!/usr/bin/env python3
import json
import os
import sys

STATE_FILE = "/tmp/usbrelay_mock_state.json"


def load_state() -> dict:
    if os.path.exists(STATE_FILE):
        try:
            with open(STATE_FILE, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            return {}
    return {}


def save_state(state: dict) -> None:
    try:
        with open(STATE_FILE, "w", encoding="utf-8") as f:
            json.dump(state, f, indent=2)
    except Exception as e:
        sys.stderr.write(f"Error saving state: {e}\n")


def main():
    args = sys.argv[1:]
    state = load_state()

    # Запрос текущего состояния всех реле (вызов без аргументов)
    if not args:
        if not state:
            state = {"959BI_1": 0}
            save_state(state)
        for relay_id, status in state.items():
            print(f"{relay_id}={status}")
        return

    # Изменение или чтение конкретного реле
    for arg in args:
        if "=" in arg:
            relay_id, val = arg.split("=", 1)
            try:
                state[relay_id] = int(val)
            except ValueError:
                pass
        else:
            relay_id = arg
            status = state.get(relay_id, 0)
            print(f"{relay_id}={status}")

    save_state(state)


if __name__ == "__main__":
    main()
